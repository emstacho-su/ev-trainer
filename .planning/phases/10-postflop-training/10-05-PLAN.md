    ```typescript
    import type { SolverNodeOutput } from "@/lib/engine/solverAdapter";
    import type { HeroPosition } from "@/lib/engine/postflopSpot";
    import type { Street } from "@/lib/engine/types";

    export interface PostflopClientRequest {
      board: string[];
      street: Street;
      potBb: number;
      effectiveStackBb: number;
      heroPosition: HeroPosition;
      history: string[];
      heroHand: string[];
    }

    const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:4000";

    export async function solvePostflopSpotClient(req: PostflopClientRequest): Promise<SolverNodeOutput> {
      const res = await fetch(`${API_BASE}/api/postflop/solve`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(req),
      });

      if (!res.ok) {
        const err = await res.json().catch(() => ({}));
        throw new Error(err?.error?.message ?? `Solver request failed: ${res.status}`);
      }

      return res.json() as Promise<SolverNodeOutput>;
    }
    ```
