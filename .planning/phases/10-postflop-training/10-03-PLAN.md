<feature>
  <name>Multi-Street Postflop State Machine</name>
  <files>src/lib/engine/postflopStateMachine.ts, src/lib/engine/postflopStateMachine.test.ts</files>
  <behavior>
    postflopReducer(state: PostflopState, action: PostflopAction) -> PostflopState

    States:
    - phase: "idle" | "flop-hero" | "flop-villain" | "turn-dealing" | "turn-hero" | "turn-villain" | "river-dealing" | "river-hero" | "river-villain" | "hand-complete"
    - offSolverLine: boolean (once true, never goes back to false)
    - streetDecisions: StreetDecision[] (accumulated per street)
    - currentBoard: string[] (grows from 3 to 4 to 5 as streets advance)
    - turnCard: string | null
    - riverCard: string | null

    Actions:
    - { type: "START_HAND", payload: { spot: PostflopSpot } } -> phase: "flop-hero"
    - { type: "HERO_ACTION", payload: { actionId: string; isOffLine: boolean } } -> phase: "flop-villain"
    - { type: "VILLAIN_ACTION", payload: { actionId: string } } -> check if street complete
      - If fold: phase: "hand-complete"
      - If street continues: phase back to "flop-hero"
      - If street complete (check-check or call): phase: "turn-dealing"
    - { type: "DEAL_TURN", payload: { card: string } } -> phase: "turn-hero", append card to board
    - { type: "DEAL_RIVER", payload: { card: string } } -> phase: "river-hero", append card to board
    - { type: "COMPLETE_HAND" } -> phase: "hand-complete"
    - { type: "RESET" } -> initial idle state

    Cases:
    - START_HAND sets phase to "flop-hero" and stores spot
    - HERO_ACTION with isOffLine:true sets offSolverLine:true and stays true on future actions
    - HERO_ACTION -> VILLAIN_ACTION -> street complete -> DEAL_TURN advances correctly
    - Fold in villain action -> hand-complete immediately
    - offSolverLine: false + HERO_ACTION(isOffLine:false) -> offSolverLine stays false
    - offSolverLine: true + HERO_ACTION(isOffLine:false) -> offSolverLine stays true
  </behavior>
  <implementation>
    Pure reducer function — no side effects, no async.
    PostflopState interface with all fields above.
    PostflopAction discriminated union type.
    createInitialState() returns idle state.
    StreetDecision: { street: Street; actionId: string; isOffLine: boolean; grade?: DecisionGrade }
    Track isStreetComplete as helper: after check-check or call action, return true.
    Street progression: FLOP -> TURN -> RIVER -> HandComplete.
    On "turn-dealing" -> "turn-hero" and "river-dealing" -> "river-hero", board is extended.
  </implementation>
</feature>

    ```typescript
    import { useReducer, useCallback } from "react";

    export function usePostflopStateMachine() {
      const [state, dispatch] = useReducer(postflopReducer, createInitialState());

      const startHand = useCallback((spot: PostflopSpot) =>
        dispatch({ type: "START_HAND", payload: { spot } }), []);

      const heroAction = useCallback((actionId: string, isOffLine: boolean) =>
        dispatch({ type: "HERO_ACTION", payload: { actionId, isOffLine } }), []);

      const villainAction = useCallback((actionId: string) =>
        dispatch({ type: "VILLAIN_ACTION", payload: { actionId } }), []);

      const dealTurn = useCallback((card: string) =>
        dispatch({ type: "DEAL_TURN", payload: { card } }), []);

      const dealRiver = useCallback((card: string) =>
        dispatch({ type: "DEAL_RIVER", payload: { card } }), []);

      const completeHand = useCallback(() =>
        dispatch({ type: "COMPLETE_HAND" }), []);

      const reset = useCallback(() =>
        dispatch({ type: "RESET" }), []);

      return { state, startHand, heroAction, villainAction, dealTurn, dealRiver, completeHand, reset };
    }
    ```
