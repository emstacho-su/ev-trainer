    ```typescript
    import type { SolverNodeOutput } from "./solverAdapter";
    import type { HeroPosition } from "./postflopSpot";
    import type { Street } from "./types";
    import { solvePostflopSubgame } from "../solver/postflopSolver";
    import { DEFAULT_CFR_CONFIG } from "../solver/cfr";

    export interface PostflopSolveRequest {
      board: string[];           // 3-5 card strings
      street: Street;
      potBb: number;
      effectiveStackBb: number;
      heroPosition: HeroPosition;
      history: string[];         // action history within the street (empty at street start)
      heroHand: string[];        // 2 hole cards
    }

    export function solvePostflopSpot(req: PostflopSolveRequest): SolverNodeOutput {
      // Use mock in development for speed; real solver in production
      if (process.env.NODE_ENV !== "production") {
        return mockSolvePostflop(req);
      }

      const config = {
        ...DEFAULT_CFR_CONFIG,
        maxIterations: 50,
        street: req.street,
        board: req.board as any, // Card type from solver
        heroRange: [],
        villainRange: [],
        potBb: req.potBb,
        stackBb: req.effectiveStackBb,
        heroPosition: req.heroPosition,
        actionAbstraction: {
          betSizes: [0.33, 0.50, 0.75, 1.0],
          raiseSizes: [2.2, 2.5, 3.0],
          allinThreshold: 1.5,
        },
      };

      const solution = solvePostflopSubgame(config as any);

      // For now return the first node's strategy (simplified)
      return {
        nodeId: `postflop:${req.street}:${req.board.join("")}`,
        actions: buildDefaultActions(req.street),
        status: solution.result.converged ? "ok" : "unsolved",
        units: "bb",
        exploitability: solution.result.exploitability,
      };
    }
    ```

    ```typescript
    import type { Request, Response } from "express";
    import { solvePostflopSpot } from "../../lib/engine/postflopSolverBridge";
    import { z } from "zod";

    const SolveRequestSchema = z.object({
      board: z.array(z.string()).min(3).max(5),
      street: z.enum(["FLOP", "TURN", "RIVER"]),
      potBb: z.number().positive(),
      effectiveStackBb: z.number().positive(),
      heroPosition: z.enum(["IP", "OOP"]),
      history: z.array(z.string()).default([]),
      heroHand: z.array(z.string()).length(2),
    });

    export async function solvePostflop(req: Request, res: Response): Promise<void> {
      const parsed = SolveRequestSchema.safeParse(req.body);
      if (!parsed.success) {
        res.status(400).json({ error: { code: "INVALID_ARGUMENT", message: parsed.error.message } });
        return;
      }
      const output = solvePostflopSpot(parsed.data);
      res.status(200).json(output);
    }
    ```
