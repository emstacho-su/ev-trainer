    ```typescript
    import type { Street, Position } from "./types";
    import type { BoardTexture } from "./boardTexture";
    import { classifyBoardTexture } from "./boardTexture";

    export type HeroPosition = "IP" | "OOP";
    export type HeroRole = "aggressor" | "caller";
    export type PreflopActionType = "srp" | "3bet" | "4bet";

    export interface PostflopSpot {
      spotId: string;
      street: Street;
      board: string[];         // 3 cards for flop, 4 for turn, 5 for river
      potBb: number;
      effectiveStackBb: number;
      heroPosition: HeroPosition;
      heroRole: HeroRole;
      heroHand: string[];      // 2 hole cards e.g. ["Ah","Kd"]
      preflopActionType: PreflopActionType;
      preflopHistory: string[]; // e.g. ["UTG", "BTN", "CALL"] auto-played preflop sequence
      texture: BoardTexture;
    }

    export interface SpotContextInfo {
      street: Street;
      heroPosition: HeroPosition;
      heroRole: HeroRole;
      preflopActionType: PreflopActionType;
    }

    const RANKS = "23456789TJQKA".split("");
    const SUITS = ["h","d","c","s"];

    export function generatePostflopBoard(excludeCards: string[] = []): string[] {
      // Generate all 52 cards, exclude specified, shuffle, take 3
      const excludeSet = new Set(excludeCards);
      const deck: string[] = [];
      for (const rank of RANKS) {
        for (const suit of SUITS) {
          const card = rank + suit;
          if (!excludeSet.has(card)) deck.push(card);
        }
      }
      // Fisher-Yates shuffle (deterministic enough for training use)
      for (let i = deck.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [deck[i], deck[j]] = [deck[j], deck[i]];
      }
      return deck.slice(0, 3);
    }

    export function deriveSpotContextLabel(info: SpotContextInfo): string {
      const prefix = info.preflopActionType === "srp" ? "SRP" :
                     info.preflopActionType === "3bet" ? "3b" : "4b";
      const pos = info.heroPosition;
      const role = info.heroRole === "aggressor" ? "Aggressor" : "Caller";
      const streetSuffix = info.street === "FLOP" ? "" :
                           info.street === "TURN" ? " (Turn)" : " (River)";
      return `${prefix} ${pos} ${role}${streetSuffix}`;
    }
    ```
