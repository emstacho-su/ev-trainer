  <action>
Create ThemeProvider client component wrapping next-themes:

**src/app/providers/ThemeProvider.tsx:**
```typescript
'use client';

import { ThemeProvider as NextThemesProvider } from 'next-themes';
import { type ThemeProviderProps } from 'next-themes/dist/types';

export function ThemeProvider({ children, ...props }: ThemeProviderProps) {
  return <NextThemesProvider {...props}>{children}</NextThemesProvider>;
}
```

Create cn() utility helper for conditional class merging:

**src/lib/utils.ts:**
```typescript
import { clsx, type ClassValue } from 'clsx';
import { twMerge } from 'tailwind-merge';

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}
```

Both files follow established patterns from RESEARCH.md. ThemeProvider is a simple re-export wrapper (Next.js convention). cn() helper combines clsx (conditional classes) with tailwind-merge (deduplication).
