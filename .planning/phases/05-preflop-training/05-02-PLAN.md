```typescript
function actionHistoryToText(history: string[]): string {
  if (history.length === 0) return 'First to act';

  const actions = history.map(id => {
    // ActionId format: 'FOLD', 'CALL', 'BET_2.5BB', 'RAISE_3.0BB'
    const [type, size] = id.split('_');
    if (size) return `${type} ${size}`;
    return type;
  });

  return actions.join(' → ');
}
```

```typescript
function getFrequencyColor(frequency: number | undefined, isUserChoice: boolean): string {
  if (isUserChoice) return 'ring-2 ring-blue-500';  // User's choice highlighted
  if (frequency === undefined) return '';             // No coloring pre-reveal

  // Frequency-weighted coloring (from RESEARCH.md Pattern 3)
  if (frequency >= 0.6) return 'bg-green-600';       // Highest-frequency solver action
  if (frequency > 0.0) return 'bg-yellow-600';        // Lower-frequency solver action
  return 'bg-red-600';                                // Non-solver action (0% frequency)
}
```

**Update button className** to include frequency-based color:

```tsx
)}
```
