    Rewrite `src/components/poker/molecules/ActionButton.tsx` to use Motion animations. Preserve all existing props and behavior — only the rendering changes.

    **Architecture:**
    - Convert the `<button>` to `<motion.button>` for tap pulse animation
    - Add a `<motion.div>` inside the button for the background color transition (since Motion can animate backgroundColor but the button itself already has Tailwind classes — use a layered approach)
    - Convert the EV `<span>` to `<motion.span>` with AnimatePresence for slide-up/fade-in

    **Detailed implementation:**

    ```tsx
    'use client';

    import { motion, AnimatePresence } from 'motion/react';
    import { cn } from '@/lib/utils';
    import { ANIM, EASE } from '@/lib/ui/animationTiming';

    // ... (keep all existing type definitions and helper functions)

    export function ActionButton({ action, label, state, ev, frequency, isUserChoice, onClick, className }: ActionButtonProps) {
      const defaultLabel = action.charAt(0).toUpperCase() + action.slice(1);
      const displayLabel = label ?? defaultLabel;
      const isRevealed = state.startsWith('revealed');
      const isCorrect = state === 'revealed-correct';
      const isSelected = state === 'selected';

      // Determine reveal background color for motion animation
      // Only used when isRevealed is true
      const revealBg = isCorrect ? '#16a34a' : '#dc2626'; // green-700 / red-600

      return (
        <div className={cn('flex flex-col gap-1.5', className)}>
          <motion.button
            onClick={onClick}
            disabled={state === 'disabled' || isRevealed}
            // Tap pulse animation — brief scale on press
            whileTap={state === 'idle' ? { scale: 0.97 } : undefined}
            // Selection pulse: when state transitions to 'selected', show a glow ring burst
            animate={isSelected ? { scale: [1, 1.04, 1] } : { scale: 1 }}
            transition={isSelected
              ? { duration: ANIM.BUTTON_PULSE, ease: EASE.IN_OUT, times: [0, 0.5, 1] }
              : { duration: 0.15 }
            }
            className={cn(
              'relative px-6 py-3 rounded-lg font-bold text-lg text-white overflow-hidden',
              'focus:outline-none focus:ring-2 focus:ring-offset-2',
              state === 'idle' && 'bg-gray-700 hover:bg-gray-600 focus:ring-gray-500',
              state === 'disabled' && 'bg-gray-800 text-gray-500 cursor-not-allowed',
              state === 'selected' && 'bg-blue-600 ring-2 ring-blue-400',
              // Revealed states: set a base bg via Tailwind, but Motion overlay will cover it
              isRevealed && 'bg-gray-700',
              isRevealed && isUserChoice && 'ring-2 ring-blue-400',
            )}
          >
            {/* Animated background overlay for reveal color transition */}
            <AnimatePresence>
              {isRevealed && (
                <motion.div
                  key="reveal-bg"
                  className="absolute inset-0 rounded-lg"
                  initial={{ opacity: 0 }}
                  animate={{ opacity: 1, backgroundColor: revealBg }}
                  exit={{ opacity: 0 }}
                  transition={{ duration: ANIM.BUTTON_COLOR, ease: EASE.OUT }}
                  style={{ zIndex: 0 }}
                />
              )}
            </AnimatePresence>

            {/* Button content (above the bg overlay) */}
            <span className="relative z-10 flex items-center gap-2">
              <span>{displayLabel}</span>
              <AnimatePresence>
                {isRevealed && ev !== undefined && (
                  <motion.span
                    key="ev-value"
                    className="text-sm font-normal"
                    initial={{ opacity: 0, y: 8 }}
                    animate={{ opacity: 1, y: 0 }}
                    exit={{ opacity: 0, y: 8 }}
                    transition={{ duration: ANIM.EV_REVEAL, ease: EASE.OUT }}
                  >
                    {ev > 0 ? '+' : ''}{ev.toFixed(2)} BB
                  </motion.span>
                )}
              </AnimatePresence>
            </span>
          </motion.button>

          {/* Frequency bar — renders instantly, no animation per spec */}
          {isRevealed && frequency !== undefined && (
            <div className="w-full">
              <div className="relative w-full h-3 bg-gray-700 rounded-full overflow-hidden">
                <div
                  className={cn('absolute left-0 top-0 h-full rounded-full', getBarColor(frequency))}
                  style={{ width: `${Math.max(frequency * 100, 2)}%` }}
                />
              </div>
              <div className="text-xs text-gray-300 text-center mt-0.5 font-medium">
                {(frequency * 100).toFixed(0)}%
              </div>
            </div>
          )}
        </div>
      );
    }
    ```

    KEY NOTES:
    - Add `'use client';` directive at top since we're using hooks/motion
    - Keep all existing helper functions (`getFrequencyBg`, `getBarColor`, type definitions)
    - The `transition-all` CSS class on the original button should be REMOVED since Motion handles transitions now (having both causes conflicts)
    - The background color animation uses a layered `motion.div` overlay approach so Tailwind's static bg classes can coexist during non-revealed states
    - `AnimatePresence` is required for `exit` animations to work (EV value fade-out when going to next hand)
    - The `getFrequencyBg` function is no longer needed for the button itself (Motion handles color), but keep it for potential future use or just remove it — `getBarColor` is still needed for the frequency bar
    - `isRevealed && getFrequencyBg(frequency)` Tailwind class is removed from the button since Motion's overlay handles reveal coloring
