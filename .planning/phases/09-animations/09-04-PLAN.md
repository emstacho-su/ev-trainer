    **Step B: Create `src/lib/audio/audioManager.ts`**

    Use individual file loading (not audio sprites) per research recommendation for simplicity with 6 files.

    ```typescript
    import { SOUNDS, type SoundName } from './sounds';

    /**
     * Web Audio API-based sound effect manager.
     *
     * Usage:
     *   await audioManager.init();  // Call on first user gesture
     *   audioManager.playSound('card-deal');
     *
     * Degrades gracefully: if audio files are missing or browser blocks audio,
     * all playSound() calls are silent (no errors thrown to the caller).
     */
    export class AudioManager {
      private audioContext: AudioContext | null = null;
      private buffers: Map<SoundName, AudioBuffer> = new Map();
      private enabled = true;
      private initialized = false;
      private initPromise: Promise<void> | null = null;

      /** Initialize audio context and begin loading sound files. Must be called from a user gesture. */
      async init(): Promise<void> {
        if (this.initialized) {
          // Resume if suspended (browser may suspend after inactivity)
          if (this.audioContext?.state === 'suspended') {
            await this.audioContext.resume();
          }
          return;
        }

        if (this.initPromise) return this.initPromise;

        this.initPromise = this._doInit();
        return this.initPromise;
      }

      private async _doInit(): Promise<void> {
        try {
          this.audioContext = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();

          if (this.audioContext.state === 'suspended') {
            await this.audioContext.resume();
          }

          // Load all sound files in parallel; failures are silent
          await Promise.allSettled(
            (Object.entries(SOUNDS) as Array<[SoundName, string]>).map(([name, url]) =>
              this._loadSound(name, url)
            )
          );

          this.initialized = true;
        } catch {
          // Audio context creation failed (e.g., restricted environment) — remain silent
        }
      }

      private async _loadSound(name: SoundName, url: string): Promise<void> {
        if (!this.audioContext) return;
        try {
          const response = await fetch(url);
          if (!response.ok) return; // File missing — degrade gracefully
          const arrayBuffer = await response.arrayBuffer();
          const audioBuffer = await this.audioContext.decodeAudioData(arrayBuffer);
          this.buffers.set(name, audioBuffer);
        } catch {
          // File missing or decode error — degrade gracefully, no console noise
        }
      }

      /** Play a named sound effect. Silent if audio is disabled, not initialized, or file missing. */
      playSound(name: SoundName, volume = 0.7): void {
        if (!this.enabled || !this.audioContext || !this.initialized) return;

        const buffer = this.buffers.get(name);
        if (!buffer) return; // File not loaded — silent

        try {
          const source = this.audioContext.createBufferSource();
          const gainNode = this.audioContext.createGain();
          source.buffer = buffer;
          gainNode.gain.setValueAtTime(volume, this.audioContext.currentTime);
          source.connect(gainNode);
          gainNode.connect(this.audioContext.destination);
          source.start(0);
        } catch {
          // Playback error — silent
        }
      }

      setEnabled(enabled: boolean): void {
        this.enabled = enabled;
      }

      isEnabled(): boolean {
        return this.enabled;
      }
    }

    /** Singleton AudioManager instance shared across the app. */
    export const audioManager = new AudioManager();
    ```

    ```typescript
    'use client';

    import { useState, useEffect, useCallback } from 'react';
    import { audioManager } from '@/lib/audio/audioManager';
    import type { SoundName } from '@/lib/audio/sounds';

    const STORAGE_KEY = 'ev-trainer-audio-enabled';

    export interface AudioControls {
      /** Whether audio is currently enabled */
      audioEnabled: boolean;
      /** Toggle audio on/off and persist to localStorage */
      toggleAudio: () => void;
      /**
       * Play a sound effect.
       * Initializes audio context on first call (satisfies browser autoplay policy).
       * Silent if audio disabled or files not loaded.
       */
      playSound: (name: SoundName, volume?: number) => void;
    }

    export function useAudio(): AudioControls {
      const [audioEnabled, setAudioEnabled] = useState(true);

      useEffect(() => {
        // Load persisted preference
        if (typeof localStorage !== 'undefined') {
          const stored = localStorage.getItem(STORAGE_KEY);
          if (stored !== null) {
            const enabled = stored === 'true';
            setAudioEnabled(enabled);
            audioManager.setEnabled(enabled);
          }
        }
      }, []);

      const toggleAudio = useCallback(() => {
        setAudioEnabled(prev => {
          const next = !prev;
          audioManager.setEnabled(next);
          if (typeof localStorage !== 'undefined') {
            localStorage.setItem(STORAGE_KEY, String(next));
          }
          return next;
        });
      }, []);

      const playSound = useCallback((name: SoundName, volume?: number) => {
        // Initialize audio context on first call (must be inside user gesture for autoplay policy).
        // init() is idempotent after first call.
        void audioManager.init().then(() => {
          audioManager.playSound(name, volume);
        });
      }, []);

      return { audioEnabled, toggleAudio, playSound };
    }
    ```
