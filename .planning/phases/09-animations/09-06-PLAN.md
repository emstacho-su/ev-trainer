    {dealerOffsets[dealerPosition] && (
      <motion.div
        layoutId="dealer-button"

    // Reusable Motion animation variants for modal dialogs.

    import type { Variants } from 'motion/react';
    import { TIMING, EASING } from './animationTiming';

    /** Backdrop/overlay fade in/out */
    export const MODAL_OVERLAY_VARIANTS: Variants = {
      hidden: { opacity: 0 },
      visible: { opacity: 1 },
    };

    /** Modal panel: fade + scale from 95% to 100% (ease-out, natural feel) */
    export const MODAL_PANEL_VARIANTS: Variants = {
      hidden: {
        opacity: 0,
        scale: 0.95,
      },
      visible: {
        opacity: 1,
        scale: 1,
      },
      exit: {
        opacity: 0,
        scale: 0.95,
      },
    };

    /** Transition config for modal open */
    export const MODAL_OPEN_TRANSITION = {
      duration: TIMING.MODAL_OPEN,
      ease: EASING.DEFAULT,
    } as const;

    /** Transition config for modal close (faster than open) */
    export const MODAL_CLOSE_TRANSITION = {
      duration: TIMING.MODAL_CLOSE,
      ease: EASING.DEFAULT,
    } as const;

    /**
     * Usage example (for Phase 7):

     *         animate="visible"
     *         exit="hidden"
     *         transition={MODAL_OPEN_TRANSITION}
     *         className="fixed inset-0 bg-black/50"

     *         className="fixed ... modal-panel"
