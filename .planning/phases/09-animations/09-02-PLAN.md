    Update `src/components/poker/atoms/Card.tsx` to export an `AnimatedCard` component that wraps the existing `Card` (face-up) and `CardBack` (face-down) in a 3D flip container.

    The existing `Card` component stays unchanged. Add a new `AnimatedCard` export below it:

    ```typescript
    import { motion } from 'motion/react';
    import { TIMING, EASING } from '@/utils/animationTiming';
    import { CardBack } from './CardBack';

    interface AnimatedCardProps {
      rank: string;
      suit: 'h' | 'd' | 'c' | 's';
      size?: 'sm' | 'md' | 'lg';
      isFlipped?: boolean; // true = show face-up (after flip), false = show face-down
      className?: string;
    }

    export function AnimatedCard({ rank, suit, size = 'md', isFlipped = false, className }: AnimatedCardProps) {
      // Size dimensions must match Card and CardBack pixel sizes
      const sizeMap = {
        sm: { width: 40, height: 56 },   // w-10 h-14
        md: { width: 56, height: 80 },   // w-14 h-20
        lg: { width: 72, height: 104 },  // w-18 h-26
      };
      const dims = sizeMap[size];

      return (
        <div
          style={{
            perspective: '1000px',
            width: dims.width,
            height: dims.height,
          }}
          className={className}
        >
          <motion.div
            animate={{ rotateY: isFlipped ? 180 : 0 }}
            transition={{ duration: TIMING.CARD_FLIP, ease: EASING.DEFAULT }}
            style={{
              width: '100%',
              height: '100%',
              position: 'relative',
              transformStyle: 'preserve-3d',
            }}
          >
            {/* Face-down side (visible when rotateY = 0) */}
            <div
              style={{
                position: 'absolute',
                width: '100%',
                height: '100%',
                backfaceVisibility: 'hidden',
              }}
            >
              <CardBack size={size} />
            </div>

            {/* Face-up side (visible when rotateY = 180) */}
            <div
              style={{
                position: 'absolute',
                width: '100%',
                height: '100%',
                backfaceVisibility: 'hidden',
                transform: 'rotateY(180deg)',
              }}
            >
              <Card rank={rank} suit={suit} size={size} />
            </div>
          </motion.div>
        </div>
      );
    }
    ```

    Important: `transformStyle: 'preserve-3d'` must be on the motion.div (the rotating element), not the outer perspective container. `backfaceVisibility: 'hidden'` must be on BOTH face divs.

    Verify: `npx tsc --noEmit` passes.
