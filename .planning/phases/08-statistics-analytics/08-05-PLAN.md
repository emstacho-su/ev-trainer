```typescript
'use client';

import { useEffect, useState } from 'react';
import { useSearchParams } from 'next/navigation';
import { format } from 'date-fns';
import type { SessionSummary } from '@/lib/stats/types';

type SortField = 'createdAt' | 'accuracy' | 'avgEVLoss' | 'decisionsCompleted';
type SortOrder = 'asc' | 'desc';

interface SessionDetail extends SessionSummary {
  entries?: Array<{
    id: string;
    actionId: string;
    result: any;
    createdAt: Date;
  }>;
  biggestMistakes?: Array<{
    entryId: string;
    evLoss: number;
  }>;
}

export default function SessionHistory() {
  const searchParams = useSearchParams();
  const [sessions, setSessions] = useState<SessionSummary[]>([]);
  const [sessionDetails, setSessionDetails] = useState<Map<string, SessionDetail>>(new Map());
  const [total, setTotal] = useState(0);
  const [page, setPage] = useState(1);
  const [pageSize] = useState(20);
  const [sortField, setSortField] = useState<SortField>('createdAt');
  const [sortOrder, setSortOrder] = useState<SortOrder>('desc');
  const [expandedSessions, setExpandedSessions] = useState<Set<string>>(new Set());
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchSessions() {
      setLoading(true);
      const startDate = searchParams.get('startDate') || new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
      const endDate = searchParams.get('endDate') || new Date().toISOString();

      const token = localStorage.getItem('accessToken');
      const response = await fetch(
        `/api/stats/sessions?startDate=${startDate}&endDate=${endDate}&page=${page}&pageSize=${pageSize}`,
        { headers: { Authorization: `Bearer ${token}` } }
      );

      const data = await response.json();

      // Client-side sorting (TODO: move to API)
      const sorted = (data.sessions || []).sort((a: SessionSummary, b: SessionSummary) => {
        const aVal = a[sortField];
        const bVal = b[sortField];
        const order = sortOrder === 'asc' ? 1 : -1;

        if (sortField === 'createdAt') {
          return order * (new Date(aVal as Date).getTime() - new Date(bVal as Date).getTime());
        }
        return order * ((aVal as number) - (bVal as number));
      });

      setSessions(sorted);
      setTotal(data.total || 0);
      setLoading(false);
    }

    fetchSessions();
  }, [searchParams, page, pageSize, sortField, sortOrder]);

  const toggleSort = (field: SortField) => {
    if (sortField === field) {
      setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc');
    } else {
      setSortField(field);
      setSortOrder('desc');
    }
  };

  const toggleExpand = async (sessionId: string) => {
    const isExpanding = !expandedSessions.has(sessionId);

    if (isExpanding && !sessionDetails.has(sessionId)) {
      // Fetch session detail with entries and biggestMistakes
      const token = localStorage.getItem('accessToken');
      const response = await fetch(`/api/stats/sessions/${sessionId}`, {
        headers: { Authorization: `Bearer ${token}` },
      });

      if (response.ok) {
        const detail = await response.json();
        setSessionDetails((prev) => new Map(prev).set(sessionId, detail));
      }
    }

    setExpandedSessions((prev) => {
      const newSet = new Set(prev);
      if (newSet.has(sessionId)) {
        newSet.delete(sessionId);
      } else {
        newSet.add(sessionId);
      }
      return newSet;
    });
  };

  const handleDelete = async (sessionId: string) => {
    if (!confirm('Delete this session? This will remove it from all statistics.')) {
      return;
    }

    const token = localStorage.getItem('accessToken');
    await fetch(`/api/stats/sessions/${sessionId}`, {
      method: 'DELETE',
      headers: { Authorization: `Bearer ${token}` },
    });

    // Refresh list
    setSessions(sessions.filter((s) => s.id !== sessionId));
  };

  const isRecent = (date: Date) => {
    return (new Date().getTime() - new Date(date).getTime()) < 24 * 60 * 60 * 1000;
  };

  if (loading) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="h-96 flex items-center justify-center text-slate-400">
          Loading session history...
        </div>
      </div>
    );
  }

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
      <h2 className="text-xl font-bold text-slate-100 mb-4">Session History</h2>

      {sessions.length === 0 ? (
        <div className="text-center text-slate-400 py-8">
          No sessions found for selected date range
        </div>
      ) : (
        <>
          {/* Table Header */}
          <div className="grid grid-cols-6 gap-4 pb-2 mb-4 border-b border-slate-700 text-sm text-slate-400">
            <button
              onClick={() => toggleSort('createdAt')}
              className="text-left hover:text-slate-200"
            >
              Date {sortField === 'createdAt' && (sortOrder === 'asc' ? '↑' : '↓')}
            </button>
            <div>Hands</div>
            <button
              onClick={() => toggleSort('accuracy')}
              className="text-left hover:text-slate-200"
            >
              Accuracy {sortField === 'accuracy' && (sortOrder === 'asc' ? '↑' : '↓')}
            </button>
            <button
              onClick={() => toggleSort('avgEVLoss')}
              className="text-left hover:text-slate-200"
            >
              Avg EV Loss {sortField === 'avgEVLoss' && (sortOrder === 'asc' ? '↑' : '↓')}
            </button>
            <div>Duration</div>
            <div className="text-right">Actions</div>
          </div>

          {/* Session Rows */}
          <div className="space-y-2">
            {sessions.map((session) => {
              const recent = isRecent(session.createdAt);
              const expanded = expandedSessions.has(session.id);
              const detail = sessionDetails.get(session.id);

              return (
                <div
                  key={session.id}
                  className={`bg-slate-800 border border-slate-700 rounded p-4 ${
                    recent ? 'border-blue-500' : ''
                  }`}
                >
                  <div className="grid grid-cols-6 gap-4 items-center text-sm">
                    <div className="font-mono text-slate-200">
                      {format(new Date(session.createdAt), 'MMM d, HH:mm')}
                      {recent && <span className="ml-2 text-xs text-blue-400">recent</span>}
                    </div>
                    <div className="text-slate-300">{session.decisionsCompleted}</div>
                    <div className="text-slate-300">{session.accuracy.toFixed(1)}%</div>
                    <div className="text-slate-300">{session.avgEVLoss.toFixed(2)} BB</div>
                    <div className="text-slate-300">{Math.floor(session.duration / 60)}m</div>
                    <div className="text-right space-x-2">
                      <button
                        onClick={() => toggleExpand(session.id)}
                        className="text-blue-500 hover:text-blue-400 text-xs"
                      >
                        {expanded ? 'Collapse' : 'Expand'}
                      </button>
                      <button
                        onClick={() => handleDelete(session.id)}
                        className="text-red-500 hover:text-red-400 text-xs"
                      >
                        Delete
                      </button>
                    </div>
                  </div>

                  {expanded && detail && (
                    <div className="mt-4 pt-4 border-t border-slate-700">
                      <div className="text-sm font-semibold text-slate-300 mb-3">Biggest Mistakes:</div>
                      {detail.biggestMistakes && detail.biggestMistakes.length > 0 ? (
                        <div className="space-y-2">
                          {detail.biggestMistakes.map((mistake, index) => (
                            <div
                              key={mistake.entryId}
                              className="bg-slate-900 border border-slate-700 rounded p-3 text-xs"
                            >
                              <div className="flex justify-between items-center">
                                <span className="text-slate-400">
                                  #{index + 1} - Entry {mistake.entryId.slice(0, 8)}...
                                </span>
                                <span className="text-red-400 font-semibold">
                                  -{mistake.evLoss.toFixed(2)} BB
                                </span>
                              </div>
                            </div>
                          ))}
                        </div>
                      ) : (
                        <div className="text-xs text-slate-500">No mistakes recorded in this session (100% accuracy)</div>
                      )}
                    </div>
                  )}

                  {expanded && !detail && (
                    <div className="mt-4 pt-4 border-t border-slate-700">
                      <div className="text-sm text-slate-400 animate-pulse">Loading details...</div>
                    </div>
                  )}
                </div>
              );
            })}
          </div>

          {/* Pagination Controls */}
          <div className="flex justify-between items-center mt-6 pt-4 border-t border-slate-700">
            <div className="text-sm text-slate-400">
              Showing {(page - 1) * pageSize + 1}-{Math.min(page * pageSize, total)} of {total} sessions
            </div>
            <div className="space-x-2">
              <button
                onClick={() => setPage(page - 1)}
                disabled={page === 1}
                className="px-4 py-2 bg-slate-800 hover:bg-slate-700 text-slate-200 rounded text-sm disabled:opacity-50 disabled:cursor-not-allowed"
              >
                Previous
              </button>
              <button
                onClick={() => setPage(page + 1)}
                disabled={page * pageSize >= total}
                className="px-4 py-2 bg-slate-800 hover:bg-slate-700 text-slate-200 rounded text-sm disabled:opacity-50 disabled:cursor-not-allowed"
              >
                Next
              </button>
            </div>
          </div>
        </>
      )}
    </div>
  );
}
```
