```typescript
// Add to existing exports

export async function toggleHandFlag(req: Request, res: Response) {
  const { sessionId, index } = req.params;
  const userId = req.user!.userId;

  try {
    // Verify session belongs to user
    const session = await prisma.session.findFirst({
      where: { id: sessionId, userId },
    });

    if (!session) {
      return res.status(404).json({ error: 'Session not found' });
    }

    // Find the session entry by sessionId and entryIndex
    const entry = await prisma.sessionEntry.findFirst({
      where: {
        sessionId,
        entryIndex: parseInt(index),
      },
    });

    if (!entry) {
      return res.status(404).json({ error: 'Session entry not found' });
    }

    // Toggle isFlagged
    const updated = await prisma.sessionEntry.update({
      where: { id: entry.id },
      data: { isFlagged: !entry.isFlagged },
    });

    res.json({ isFlagged: updated.isFlagged });
  } catch (error) {
    console.error('Error toggling hand flag:', error);
    res.status(500).json({ error: 'Failed to toggle flag' });
  }
}
```

Create FlaggedHandsList component to show all flagged hands:

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useSearchParams } from 'next/navigation';
import { format } from 'date-fns';

interface FlaggedHand {
  sessionId: string;
  sessionCreatedAt: Date;
  entryIndex: number;
  actionId: string;
  result: any;
  isFlagged: boolean;
}

export default function FlaggedHandsList() {
  const searchParams = useSearchParams();
  const [flaggedHands, setFlaggedHands] = useState<FlaggedHand[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchFlaggedHands() {
      setLoading(true);
      const startDate = searchParams.get('startDate') || new Date(Date.now() - 365 * 24 * 60 * 60 * 1000).toISOString();
      const endDate = searchParams.get('endDate') || new Date().toISOString();

      const token = localStorage.getItem('accessToken');

      // Fetch sessions and filter for flagged entries
      const response = await fetch(
        `/api/stats/sessions?startDate=${startDate}&endDate=${endDate}&page=1&pageSize=100`,
        { headers: { Authorization: `Bearer ${token}` } }
      );

      const data = await response.json();

      // For each session, fetch entries and collect flagged ones
      const flagged: FlaggedHand[] = [];
      for (const session of data.sessions || []) {
        const sessionResponse = await fetch(`/api/session/${session.id}`, {
          headers: { Authorization: `Bearer ${token}` },
        });
        const sessionData = await sessionResponse.json();

        (sessionData.entries || []).forEach((entry: any, idx: number) => {
          if (entry.isFlagged) {
            flagged.push({
              sessionId: session.id,
              sessionCreatedAt: session.createdAt,
              entryIndex: idx,
              actionId: entry.actionId,
              result: entry.result,
              isFlagged: entry.isFlagged,
            });
          }
        });
      }

      setFlaggedHands(flagged);
      setLoading(false);
    }

    fetchFlaggedHands();
  }, [searchParams]);

  const unflagHand = async (sessionId: string, entryIndex: number) => {
    const token = localStorage.getItem('accessToken');
    await fetch(`/api/stats/sessions/${sessionId}/entries/${entryIndex}/flag`, {
      method: 'PATCH',
      headers: { Authorization: `Bearer ${token}` },
    });

    // Remove from list
    setFlaggedHands(flaggedHands.filter(
      (h) => !(h.sessionId === sessionId && h.entryIndex === entryIndex)
    ));
  };

  if (loading) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="h-64 flex items-center justify-center text-slate-400">
          Loading flagged hands...
        </div>
      </div>
    );
  }

  if (flaggedHands.length === 0) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <h2 className="text-xl font-bold text-slate-100 mb-4">Flagged Hands</h2>
        <div className="text-center text-slate-400 py-8">
          No flagged hands yet. Flag difficult spots during session review to revisit them here.
        </div>
      </div>
    );
  }

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
      <h2 className="text-xl font-bold text-slate-100 mb-4">
        Flagged Hands ({flaggedHands.length})
      </h2>

      <div className="space-y-2">
        {flaggedHands.map((hand, i) => (
          <div
            key={i}
            className="flex items-center justify-between bg-slate-800 border border-slate-700 rounded p-4"
          >
            <div className="flex-1">
              <div className="text-sm font-mono text-slate-200">
                Session {format(new Date(hand.sessionCreatedAt), 'MMM d, HH:mm')} · Hand #{hand.entryIndex + 1}
              </div>
              <div className="text-xs text-slate-400 mt-1">
                {hand.result && (
                  <>
                    {(hand.result as any).isBestAction ? '✓ Correct' : '✗ Mistake'}
                    {(hand.result as any).evLoss && (
                      <span className="ml-2">EV Loss: {((hand.result as any).evLoss).toFixed(2)} BB</span>
                    )}
                  </>
                )}
              </div>
            </div>

            <div className="flex items-center gap-2">
              <button
                onClick={() => unflagHand(hand.sessionId, hand.entryIndex)}
                className="px-3 py-1.5 bg-slate-700 hover:bg-slate-600 text-slate-200 text-xs rounded"
              >
                Unflag
              </button>
              <button
                className="px-3 py-1.5 bg-blue-600 hover:bg-blue-500 text-white text-xs rounded"
                onClick={() => {
                  // TODO: Navigate to hand review (future enhancement)
                  alert('Hand review feature coming soon');
                }}
              >
                Review
              </button>
            </div>
          </div>
        ))}
      </div>

      <div className="mt-4 text-xs text-slate-500">
        Flagged hands are saved and persist across sessions. Use them to revisit challenging spots.
      </div>
    </div>
  );
}
```
