Build Recharts line chart with metric toggle dropdown:

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useSearchParams } from 'next/navigation';
import {
  LineChart,
  Line,
  CartesianGrid,
  XAxis,
  YAxis,
  Tooltip,
  ResponsiveContainer,
  TooltipProps,
} from 'recharts';
import { format } from 'date-fns';
import type { PerformanceDataPoint } from '@/lib/stats/types';

type MetricKey = 'avgEVLoss' | 'accuracy' | 'hands' | 'correctFoldPct' | 'correctRaisePct';

const METRICS: Array<{ key: MetricKey; label: string; unit: string; color: string }> = [
  { key: 'avgEVLoss', label: 'Avg EV Loss', unit: 'BB', color: '#10b981' },
  { key: 'accuracy', label: 'Accuracy', unit: '%', color: '#3b82f6' },
  { key: 'hands', label: 'Hands Played', unit: '', color: '#8b5cf6' },
  { key: 'correctFoldPct', label: 'Correct Fold %', unit: '%', color: '#f59e0b' },
  { key: 'correctRaisePct', label: 'Correct Raise %', unit: '%', color: '#ef4444' },
];

export default function PerformanceChart() {
  const searchParams = useSearchParams();
  const [selectedMetric, setSelectedMetric] = useState<MetricKey>('avgEVLoss');
  const [data, setData] = useState<PerformanceDataPoint[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchData() {
      setLoading(true);
      const startDate = searchParams.get('startDate') || new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
      const endDate = searchParams.get('endDate') || new Date().toISOString();

      const token = localStorage.getItem('accessToken');
      const response = await fetch(
        `/api/stats/performance?startDate=${startDate}&endDate=${endDate}`,
        { headers: { Authorization: `Bearer ${token}` } }
      );

      const result = await response.json();
      setData(result.metrics || []);
      setLoading(false);
    }

    fetchData();
  }, [searchParams]);

  const CustomTooltip = ({ active, payload }: TooltipProps<number, string>) => {
    if (!active || !payload || payload.length === 0) return null;
    const dataPoint = payload[0].payload as PerformanceDataPoint;
    const metric = METRICS.find((m) => m.key === selectedMetric);

    return (
      <div className="bg-slate-950 border border-slate-700 rounded p-3 shadow-lg text-xs">
        <p className="font-mono text-slate-200 mb-1">{format(new Date(dataPoint.date), 'MMM d, yyyy')}</p>
        <p className="text-slate-100">
          <span className="font-semibold">{metric?.label}:</span>{' '}
          {dataPoint[selectedMetric]?.toFixed(2)} {metric?.unit}
        </p>
        <p className="text-slate-400 mt-1">Hands: {dataPoint.hands}</p>
        <p className="text-slate-400">Sessions: {dataPoint.sessionCount}</p>
      </div>
    );
  };

  if (loading) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="h-96 flex items-center justify-center text-slate-400">
          <div className="animate-pulse">Loading chart...</div>
        </div>
      </div>
    );
  }

  if (data.length === 0) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="h-96 flex items-center justify-center text-slate-400">
          No data for selected date range
        </div>
      </div>
    );
  }

  const currentMetric = METRICS.find((m) => m.key === selectedMetric);

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
      <div className="flex justify-between items-center mb-4">
        <h2 className="text-xl font-bold text-slate-100">Performance Over Time</h2>
        <select
          value={selectedMetric}
          onChange={(e) => setSelectedMetric(e.target.value as MetricKey)}
          className="px-3 py-2 bg-slate-800 border border-slate-700 text-slate-200 rounded text-sm"
        >
          {METRICS.map((metric) => (
            <option key={metric.key} value={metric.key}>
              {metric.label}
            </option>
          ))}
        </select>
      </div>

      <ResponsiveContainer width="100%" height={400}>
        <LineChart data={data} margin={{ top: 5, right: 30, left: 0, bottom: 5 }}>
          <CartesianGrid
            strokeDasharray="3 3"
            stroke="hsl(var(--chart-grid))"
            vertical={false}
          />
          <XAxis
            dataKey="date"
            stroke="hsl(var(--chart-axis))"
            style={{ fontSize: '12px' }}
            tickFormatter={(date) => format(new Date(date), 'MMM d')}
          />
          <YAxis
            stroke="hsl(var(--chart-axis))"
            style={{ fontSize: '12px' }}
          />
          <Tooltip content={<CustomTooltip />} cursor={{ stroke: '#64748b' }} />
          <Line
            type="monotone"
            dataKey={selectedMetric}
            stroke={currentMetric?.color || '#10b981'}
            dot={false}
            strokeWidth={2}
            isAnimationActive={false}
          />
        </LineChart>
      </ResponsiveContainer>
    </div>
  );
}
```

Follow CONTEXT.md: single metric at a time, dropdown toggle, 5 metrics available. RESEARCH.md WARNING: Hardcoding colors breaks dark theme - use CSS variables.

'use client';

import { Suspense, useState } from 'react';
import MetricCards from './components/MetricCards';
import FilterBar from './components/FilterBar';
import PerformanceChart from './components/PerformanceChart';
import { MetricCardSkeleton, ChartSkeleton } from './components/LoadingSkeletons';

export default function StatsPage() {
  const [activeTab, setActiveTab] = useState<'performance' | 'positions' | 'sessions'>('performance');

  return (
    <div className="space-y-8">
      <div>
        <h1 className="text-3xl font-bold text-slate-100 mb-2">Performance Statistics</h1>
        <p className="text-slate-400">Track your progress and identify areas for improvement</p>
      </div>

      <Suspense fallback={<div className="text-slate-400">Loading filters...</div>}>
        <FilterBar />
      </Suspense>

      <Suspense fallback={
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
          {[1, 2, 3, 4].map(i => <MetricCardSkeleton key={i} />)}
        </div>
      }>
        <MetricCards />
      </Suspense>

      {/* Tabbed sections */}
      <div className="bg-slate-900 rounded-lg border border-slate-800">
        <div className="border-b border-slate-800 flex">
          <button
            onClick={() => setActiveTab('performance')}
            className={`px-6 py-3 text-sm font-medium ${
              activeTab === 'performance'
                ? 'text-slate-100 border-b-2 border-blue-500'
                : 'text-slate-400 hover:text-slate-200'
            }`}
          >
            Performance
          </button>
          <button
            onClick={() => setActiveTab('positions')}
            className={`px-6 py-3 text-sm font-medium ${
              activeTab === 'positions'
                ? 'text-slate-100 border-b-2 border-blue-500'
                : 'text-slate-400 hover:text-slate-200'
            }`}
          >
            Positions
          </button>
          <button
            onClick={() => setActiveTab('sessions')}
            className={`px-6 py-3 text-sm font-medium ${
              activeTab === 'sessions'
                ? 'text-slate-100 border-b-2 border-blue-500'
                : 'text-slate-400 hover:text-slate-200'
            }`}
          >
            Sessions
          </button>
        </div>

          {activeTab === 'performance' && (
            <Suspense fallback={<ChartSkeleton />}>
              <PerformanceChart />
            </Suspense>
          )}
          {activeTab === 'positions' && (
            <div className="text-slate-400">Positions tab - coming in plan 04</div>
          )}
          {activeTab === 'sessions' && (
            <div className="text-slate-400">Sessions tab - coming in plan 05</div>
          )}
        </div>
      </div>
    </div>
  );
}
