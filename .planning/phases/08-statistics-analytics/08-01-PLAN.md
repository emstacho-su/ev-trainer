```typescript
import { prisma } from '@/lib/prisma';
import { startOfDay, endOfDay, eachDayOfInterval } from 'date-fns';
import type { PerformanceDataPoint, PositionStat, SessionSummary, StatsFilters } from './types';

// Adaptive granularity based on date range
function getGranularity(startDate: Date, endDate: Date): 'session' | 'day' | 'week' {
  const days = (endDate.getTime() - startDate.getTime()) / (1000 * 60 * 60 * 24);
  if (days <= 7) return 'session';
  if (days <= 30) return 'day';
  return 'week';
}

// Get daily performance stats with server-side aggregation
export async function getDailyStats(
  userId: string,
  filters: StatsFilters
): Promise<PerformanceDataPoint[]> {
  // Use Prisma groupBy on DailyStat table for efficient aggregation
  const stats = await prisma.dailyStat.groupBy({
    by: ['date'],
    where: {
      userId,
      date: {
        gte: startOfDay(filters.startDate),
        lte: endOfDay(filters.endDate),
      },
    },
    _sum: {
      totalDecisions: true,
      correctDecisions: true,
      sessionsCompleted: true,
    },
    _avg: {
      avgEvLoss: true,
    },
    orderBy: { date: 'asc' },
  });

  // Transform into chart format
  return stats.map((day) => ({
    date: day.date.toISOString().split('T')[0],
    hands: day._sum.totalDecisions ?? 0,
    accuracy: ((day._sum.correctDecisions ?? 0) / (day._sum.totalDecisions ?? 1)) * 100,
    avgEVLoss: Number(day._avg.avgEvLoss ?? 0),
    correctFoldPct: 0, // TODO: requires action-type tracking in schema
    correctRaisePct: 0, // TODO: requires action-type tracking in schema
    sessionCount: day._sum.sessionsCompleted ?? 0,
  }));
}

// Get position-based performance breakdown
export async function getPositionBreakdown(
  userId: string,
  filters: StatsFilters,
  threshold = 20
): Promise<PositionStat[]> {
  const stats = await prisma.spotStat.groupBy({
    by: ['heroPosition', 'villainPosition'],
    where: {
      userId,
      lastPracticed: {
        gte: filters.startDate,
        lte: filters.endDate,
      },
    },
    _sum: {
      totalDecisions: true,
      correctDecisions: true,
    },
    _avg: {
      avgEvLoss: true,
    },
  });

  return stats.map((stat) => ({
    heroPosition: stat.heroPosition,
    villainPosition: stat.villainPosition,
    hands: stat._sum.totalDecisions ?? 0,
    accuracy: ((stat._sum.correctDecisions ?? 0) / (stat._sum.totalDecisions ?? 1)) * 100,
    avgEVLoss: Number(stat._avg.avgEvLoss ?? 0),
    confidence: (stat._sum.totalDecisions ?? 0) >= threshold,
  }));
}

// Get session list with pagination
export async function getSessionList(
  userId: string,
  filters: StatsFilters,
  page = 1,
  pageSize = 20
): Promise<{ sessions: SessionSummary[]; total: number }> {
  const where = {
    userId,
    createdAt: {
      gte: filters.startDate,
      lte: filters.endDate,
    },
    isComplete: true,
  };

  const [sessions, total] = await Promise.all([
    prisma.session.findMany({
      where,
      include: {
        entries: {
          select: {
            actionId: true,
            result: true,
            createdAt: true,
          },
        },
      },
      orderBy: { createdAt: 'desc' },
      skip: (page - 1) * pageSize,
      take: pageSize,
    }),
    prisma.session.count({ where }),
  ]);

  return {
    sessions: sessions.map((session) => {
      const correctCount = session.entries.filter((e) =>
        e.result && typeof e.result === 'object' && 'isBestAction' in e.result
          ? (e.result as any).isBestAction
          : false
      ).length;

      const evLosses = session.entries
        .map((e) =>
          e.result && typeof e.result === 'object' && 'evLoss' in e.result
            ? Number((e.result as any).evLoss)
            : 0
        )
        .filter((loss) => loss !== 0);

      const firstEntry = session.entries[0]?.createdAt;
      const lastEntry = session.entries[session.entries.length - 1]?.createdAt;
      const duration = firstEntry && lastEntry
        ? (lastEntry.getTime() - firstEntry.getTime()) / 1000
        : 0;

      return {
        id: session.id,
        sessionId: session.sessionId,
        createdAt: session.createdAt,
        decisionsPerSession: session.decisionsPerSession,
        decisionsCompleted: session.entries.length,
        accuracy: (correctCount / session.entries.length) * 100,
        avgEVLoss: evLosses.reduce((a, b) => a + b, 0) / evLosses.length || 0,
        duration,
        scenarioBreakdown: {}, // TODO: extract from spot metadata
      };
    }),
    total,
  };
}
```

```typescript
import { Router } from 'express';
import { requireAuth } from '../auth/auth.middleware';
import { validate } from '../middleware/validation.middleware';
import { z } from 'zod';
import * as statsController from '../controllers/stats.controller';

export const statsRouter = Router();

// All stats routes require authentication
statsRouter.use(requireAuth);

// GET /api/stats/performance?startDate=...&endDate=...&positions=...&scenarios=...
const performanceQuerySchema = z.object({
  startDate: z.string().datetime(),
  endDate: z.string().datetime(),
  positions: z.string().optional(),
  scenarios: z.string().optional(),
});

statsRouter.get(
  '/performance',
  validate(performanceQuerySchema),
  statsController.getPerformanceStats
);

// GET /api/stats/positions?startDate=...&endDate=...
statsRouter.get(
  '/positions',
  validate(performanceQuerySchema),
  statsController.getPositionStats
);

// GET /api/stats/sessions?page=1&pageSize=20&startDate=...&endDate=...
const sessionHistoryQuerySchema = performanceQuerySchema.extend({
  page: z.string().optional(),
  pageSize: z.string().optional(),
});

statsRouter.get(
  '/sessions',
  validate(sessionHistoryQuerySchema),
  statsController.getSessionHistory
);
```

```typescript
import { Request, Response } from 'express';
import { getDailyStats, getPositionBreakdown, getSessionList } from '@/lib/stats/aggregation';
import type { StatsFilters } from '@/lib/stats/types';

export async function getPerformanceStats(req: Request, res: Response) {
  const { startDate, endDate, positions, scenarios } = req.query;
  const userId = req.user!.userId;

  const filters: StatsFilters = {
    startDate: new Date(startDate as string),
    endDate: new Date(endDate as string),
    positions: positions ? (positions as string).split(',') : undefined,
    scenarios: scenarios ? (scenarios as string).split(',') : undefined,
  };

  const metrics = await getDailyStats(userId, filters);

  res.json({
    metrics,
    granularity: 'day', // TODO: compute from date range
  });
}

export async function getPositionStats(req: Request, res: Response) {
  const { startDate, endDate } = req.query;
  const userId = req.user!.userId;

  const filters: StatsFilters = {
    startDate: new Date(startDate as string),
    endDate: new Date(endDate as string),
  };

  const stats = await getPositionBreakdown(userId, filters);

  res.json({
    stats,
    lowConfidenceThreshold: 20,
  });
}

export async function getSessionHistory(req: Request, res: Response) {
  const { startDate, endDate, page = '1', pageSize = '20' } = req.query;
  const userId = req.user!.userId;

  const filters: StatsFilters = {
    startDate: new Date(startDate as string),
    endDate: new Date(endDate as string),
  };

  const { sessions, total } = await getSessionList(
    userId,
    filters,
    parseInt(page as string),
    parseInt(pageSize as string)
  );

  res.json({
    sessions,
    total,
    page: parseInt(page as string),
    pageSize: parseInt(pageSize as string),
    hasMore: total > parseInt(page as string) * parseInt(pageSize as string),
  });
}
```
