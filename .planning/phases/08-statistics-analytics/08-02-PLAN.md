```typescript
'use client';

import { useRouter } from 'next/navigation';
import { useEffect, useState } from 'react';

export default function StatsLayout({ children }: { children: React.ReactNode }) {
  const router = useRouter();
  const [isAuthenticated, setIsAuthenticated] = useState(false);

  useEffect(() => {
    // Check for access token in localStorage
    const token = localStorage.getItem('accessToken');
    if (!token) {
      // Redirect to login with return URL
      router.push(`/login?returnUrl=/stats`);
    } else {
      setIsAuthenticated(true);
    }
  }, [router]);

  if (!isAuthenticated) {
    return (
      <div className="flex items-center justify-center min-h-screen bg-slate-950">
        <div className="text-slate-400">Checking authentication...</div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-950">
      <div className="container mx-auto px-4 py-8 max-w-7xl">
        {children}
      </div>
    </div>
  );
}
```

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useSearchParams } from 'next/navigation';
import { ArrowUpIcon, ArrowDownIcon } from '@heroicons/react/24/solid'; // or lucide-react

interface MetricData {
  totalHands: number;
  accuracy: number;
  avgEVLoss: number;
  currentStreak: number;
  trends: {
    totalHandsDelta: number;
    accuracyDelta: number;
    avgEVLossDelta: number;
    currentStreakDelta: number;
  };
}

export default function MetricCards() {
  const searchParams = useSearchParams();
  const [metrics, setMetrics] = useState<MetricData | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchMetrics() {
      const startDate = searchParams.get('startDate') || new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
      const endDate = searchParams.get('endDate') || new Date().toISOString();

      const token = localStorage.getItem('accessToken');
      const response = await fetch(
        `/api/stats/performance?startDate=${startDate}&endDate=${endDate}`,
        { headers: { Authorization: `Bearer ${token}` } }
      );

      const data = await response.json();

      // Compute metrics from aggregated data
      const totalHands = data.metrics.reduce((sum: number, d: any) => sum + d.hands, 0);
      const avgAccuracy = data.metrics.reduce((sum: number, d: any) => sum + d.accuracy, 0) / data.metrics.length || 0;
      const avgEVLoss = data.metrics.reduce((sum: number, d: any) => sum + d.avgEVLoss, 0) / data.metrics.length || 0;

      // TODO: Compute trends by comparing with previous period
      setMetrics({
        totalHands,
        accuracy: avgAccuracy,
        avgEVLoss,
        currentStreak: 0, // TODO: compute from session data
        trends: {
          totalHandsDelta: 0,
          accuracyDelta: 0,
          avgEVLossDelta: 0,
          currentStreakDelta: 0,
        },
      });
      setLoading(false);
    }

    fetchMetrics();
  }, [searchParams]);

  if (loading) {
    return (
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
        {[1, 2, 3, 4].map(i => (
          <div key={i} className="h-32 bg-slate-900 border border-slate-800 animate-pulse rounded-lg" />
        ))}
      </div>
    );
  }

  const cards = [
    { label: 'Total Hands', value: metrics?.totalHands.toLocaleString(), delta: metrics?.trends.totalHandsDelta || 0 },
    { label: 'Accuracy', value: `${metrics?.accuracy.toFixed(1)}%`, delta: metrics?.trends.accuracyDelta || 0 },
    { label: 'Avg EV Loss', value: `${metrics?.avgEVLoss.toFixed(2)} BB`, delta: metrics?.trends.avgEVLossDelta || 0 },
    { label: 'Current Streak', value: metrics?.currentStreak.toString() || '0', delta: metrics?.trends.currentStreakDelta || 0 },
  ];

  return (
    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
      {cards.map((card, i) => (
        <div key={i} className="bg-slate-900 border border-slate-800 rounded-lg p-6">
          <div className="text-sm text-slate-400 mb-1">{card.label}</div>
          <div className="text-3xl font-bold text-slate-100 mb-2">{card.value}</div>
          <div className="flex items-center text-sm">
            {card.delta > 0 ? (
              <ArrowUpIcon className="w-4 h-4 text-green-500 mr-1" />
            ) : card.delta < 0 ? (
              <ArrowDownIcon className="w-4 h-4 text-red-500 mr-1" />
            ) : null}
            <span className={card.delta > 0 ? 'text-green-500' : card.delta < 0 ? 'text-red-500' : 'text-slate-500'}>
              {card.delta > 0 ? '+' : ''}{card.delta.toFixed(1)}%
            </span>
            <span className="text-slate-500 ml-1">vs prev period</span>
          </div>
        </div>
      ))}
    </div>
  );
}
```

Create filter bar with shadcn/ui Calendar for date range selection and URL persistence:

```typescript
'use client';

import { useState } from 'react';
import { useSearchParams, useRouter } from 'next/navigation';
import { format, subDays } from 'date-fns';
import { Button } from '@/components/ui/button';
import { Calendar } from '@/components/ui/calendar';
import { Popover, PopoverContent, PopoverTrigger } from '@/components/ui/popover';

export default function FilterBar() {
  const router = useRouter();
  const searchParams = useSearchParams();

  const [startDate, setStartDate] = useState(
    searchParams.get('startDate')
      ? new Date(searchParams.get('startDate')!)
      : subDays(new Date(), 30)
  );
  const [endDate, setEndDate] = useState(
    searchParams.get('endDate') ? new Date(searchParams.get('endDate')!) : new Date()
  );

  const applyFilters = () => {
    const params = new URLSearchParams(searchParams);
    params.set('startDate', startDate.toISOString());
    params.set('endDate', endDate.toISOString());
    router.push(`/stats?${params.toString()}`);
  };

  const setPreset = (days: number) => {
    const end = new Date();
    const start = subDays(end, days);
    setStartDate(start);
    setEndDate(end);

    const params = new URLSearchParams(searchParams);
    params.set('startDate', start.toISOString());
    params.set('endDate', end.toISOString());
    router.push(`/stats?${params.toString()}`);
  };

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-lg p-4 sticky top-4 z-10">
      <div className="flex flex-wrap items-center gap-4">
        <div className="flex-1">
          <label className="block text-sm text-slate-400 mb-2">Date Range</label>
          <div className="flex gap-2">
            <Button
              onClick={() => setPreset(7)}
              variant="outline"
              size="sm"
            >
              7D
            </Button>
            <Button
              onClick={() => setPreset(30)}
              variant="outline"
              size="sm"
            >
              30D
            </Button>
            <Button
              onClick={() => setPreset(90)}
              variant="outline"
              size="sm"
            >
              90D
            </Button>
            <Button
              onClick={() => setPreset(365)}
              variant="outline"
              size="sm"
            >
              All
            </Button>
          </div>
        </div>

        <div className="flex-1">
          <label className="block text-sm text-slate-400 mb-2">Custom Range</label>
          <div className="flex gap-2 items-center">
            <Popover>
              <PopoverTrigger asChild>
                <Button variant="outline" size="sm">
                  {format(startDate, 'MMM d, yyyy')}
                </Button>
              </PopoverTrigger>
              <PopoverContent className="w-auto p-0 bg-slate-900 border-slate-700" align="start">
                <Calendar
                  mode="single"
                  selected={startDate}
                  onSelect={(date) => date && setStartDate(date)}
                  disabled={(date) => date > endDate}
                />
              </PopoverContent>
            </Popover>

            <span className="text-slate-500">to</span>

            <Popover>
              <PopoverTrigger asChild>
                <Button variant="outline" size="sm">
                  {format(endDate, 'MMM d, yyyy')}
                </Button>
              </PopoverTrigger>
              <PopoverContent className="w-auto p-0 bg-slate-900 border-slate-700" align="start">
                <Calendar
                  mode="single"
                  selected={endDate}
                  onSelect={(date) => date && setEndDate(date)}
                  disabled={(date) => date < startDate}
                />
              </PopoverContent>
            </Popover>

            <Button onClick={applyFilters} size="sm">
              Apply
            </Button>
          </div>
        </div>

        {/* Placeholder for position/scenario filters - can be added later */}
        <div className="flex-1 text-sm text-slate-500">
          Position and scenario filters: coming soon
        </div>
      </div>
    </div>
  );
}
```
