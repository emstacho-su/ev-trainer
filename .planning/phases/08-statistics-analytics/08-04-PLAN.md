```typescript
'use client';

import { useEffect, useState } from 'react';
import { useSearchParams } from 'next/navigation';
import clsx from 'clsx';
import type { PositionStat } from '@/lib/stats/types';

type HeatmapMetric = 'accuracy' | 'avgEVLoss';

const POSITIONS_6MAX = ['UTG', 'HJ', 'CO', 'BTN', 'SB', 'BB'];

export default function PositionHeatmap() {
  const searchParams = useSearchParams();
  const [metric, setMetric] = useState<HeatmapMetric>('accuracy');
  const [stats, setStats] = useState<PositionStat[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchStats() {
      setLoading(true);
      const startDate = searchParams.get('startDate') || new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
      const endDate = searchParams.get('endDate') || new Date().toISOString();

      const token = localStorage.getItem('accessToken');
      const response = await fetch(
        `/api/stats/positions?startDate=${startDate}&endDate=${endDate}`,
        { headers: { Authorization: `Bearer ${token}` } }
      );

      const data = await response.json();
      setStats(data.stats || []);
      setLoading(false);
    }

    fetchStats();
  }, [searchParams]);

  const getIntensityClass = (value: number, isAccuracy: boolean): string => {
    // Normalize accuracy (0-100) or EV loss (0-5+ BB)
    const normalized = isAccuracy
      ? Math.min(Math.max(value, 0), 100)
      : 100 - Math.min((value / 3) * 100, 100); // Lower EV loss = better (green)

    if (normalized < 40) return 'bg-red-900 text-slate-100';
    if (normalized < 70) return 'bg-yellow-700 text-slate-900';
    return 'bg-green-800 text-slate-100';
  };

  const getCellStat = (heroPos: string, villainPos: string | null): PositionStat | undefined => {
    return stats.find(
      (s) => s.heroPosition === heroPos && s.villainPosition === villainPos
    );
  };

  if (loading) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="h-64 flex items-center justify-center text-slate-400">
          Loading heatmap...
        </div>
      </div>
    );
  }

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
      <div className="flex justify-between items-center mb-4">
        <h2 className="text-xl font-bold text-slate-100">Position Performance</h2>
        <select
          value={metric}
          onChange={(e) => setMetric(e.target.value as HeatmapMetric)}
          className="px-3 py-2 bg-slate-800 border border-slate-700 text-slate-200 rounded text-sm"
        >
          <option value="accuracy">Accuracy %</option>
          <option value="avgEVLoss">Avg EV Loss (BB)</option>
        </select>
      </div>

      <div className="grid grid-cols-7 gap-2 mb-2">
        <div className="text-center text-xs text-slate-500 font-mono">vs →</div>
        {POSITIONS_6MAX.map((pos) => (
          <div key={pos} className="text-center text-xs text-slate-400 font-mono">
            {pos}
          </div>
        ))}
      </div>

      {POSITIONS_6MAX.map((heroPos) => (
        <div key={heroPos} className="grid grid-cols-7 gap-2 mb-2">
          <div className="text-right text-xs text-slate-400 font-mono pr-2 flex items-center justify-end">
            {heroPos}
          </div>
          {POSITIONS_6MAX.map((villainPos) => {
            const stat = getCellStat(heroPos, villainPos);
            if (!stat) {
              return (
                <div
                  key={villainPos}
                  className="bg-slate-800 rounded p-2 text-center text-xs text-slate-600"
                >
                  —
                </div>
              );
            }

            const value = metric === 'accuracy' ? stat.accuracy : stat.avgEVLoss;
            const displayValue = metric === 'accuracy' ? `${value.toFixed(0)}%` : value.toFixed(2);

            return (
              <div
                key={villainPos}
                className={clsx(
                  'rounded p-2 text-center text-xs font-mono transition-opacity',
                  getIntensityClass(value, metric === 'accuracy'),
                  !stat.confidence && 'opacity-50'
                )}
                title={`${heroPos} vs ${villainPos}: ${displayValue} (${stat.hands} hands)`}
              >
                <div className="font-bold">{displayValue}</div>
                {!stat.confidence && <div className="text-xs mt-0.5">*</div>}
              </div>
            );
          })}
        </div>
      ))}

      <div className="mt-4 text-xs text-slate-500">
        * Less than 20 hands (low confidence)
      </div>
    </div>
  );
}

'use client';

import { useEffect, useState } from 'react';
import { useSearchParams, useRouter } from 'next/navigation';
import type { PositionStat } from '@/lib/stats/types';

export default function WeaknessBreakdown() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const [stats, setStats] = useState<PositionStat[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchStats() {
      setLoading(true);
      const startDate = searchParams.get('startDate') || new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
      const endDate = searchParams.get('endDate') || new Date().toISOString();

      const token = localStorage.getItem('accessToken');
      const response = await fetch(
        `/api/stats/positions?startDate=${startDate}&endDate=${endDate}`,
        { headers: { Authorization: `Bearer ${token}` } }
      );

      const data = await response.json();
      // Sort by worst performance (lowest accuracy or highest EV loss)
      const sorted = (data.stats || [])
        .filter((s: PositionStat) => s.confidence) // Only show confident data
        .sort((a: PositionStat, b: PositionStat) => a.accuracy - b.accuracy);

      setStats(sorted.slice(0, 10)); // Top 10 weaknesses
      setLoading(false);
    }

    fetchStats();
  }, [searchParams]);

  const handleDrill = (stat: PositionStat) => {
    // Navigate to training with position filter
    const filters = {
      heroPosition: stat.heroPosition,
      villainPosition: stat.villainPosition,
    };
    router.push(`/training?filters=${encodeURIComponent(JSON.stringify(filters))}`);
  };

  if (loading) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="h-64 flex items-center justify-center text-slate-400">
          Loading weakness breakdown...
        </div>
      </div>
    );
  }

  if (stats.length === 0) {
    return (
      <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
        <div className="text-center text-slate-400 py-8">
          No weaknesses detected. Keep training!
        </div>
      </div>
    );
  }

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-lg p-6">
      <h2 className="text-xl font-bold text-slate-100 mb-4">Top Weaknesses</h2>

      <div className="space-y-2">
        {stats.map((stat, i) => (
          <div
            key={i}
            className="flex items-center justify-between bg-slate-800 border border-slate-700 rounded p-4"
          >
            <div className="flex-1">
              <div className="text-sm font-mono text-slate-200">
                {stat.heroPosition} vs {stat.villainPosition || 'All'}
              </div>
              <div className="text-xs text-slate-400 mt-1">
                {stat.hands} hands · {stat.accuracy.toFixed(1)}% accuracy · {stat.avgEVLoss.toFixed(2)} BB avg loss
              </div>
            </div>

            <button
              onClick={() => handleDrill(stat)}
              className="px-4 py-2 bg-blue-600 hover:bg-blue-500 text-white text-sm font-medium rounded"
            >
              Drill this
            </button>
          </div>
        ))}
      </div>

      <div className="mt-4 text-xs text-slate-500">
        Showing top 10 weaknesses with sufficient sample size (≥20 hands)
      </div>
    </div>
  );
}
```
